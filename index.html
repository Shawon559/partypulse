    ⚡ 1-Click Supabase SQL Setup Schema

    Paste this into your Supabase Dashboard > SQL Editor to create the tables with Realtime enabled:

    -- PartyPulse Supabase Database Schema
CREATE TABLE IF NOT EXISTS rooms (
  code TEXT PRIMARY KEY,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS songs (
  id TEXT PRIMARY KEY,
  room_code TEXT REFERENCES rooms(code) ON DELETE CASCADE,
  title TEXT NOT NULL,
  votes INT4 DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS vibes (
  id TEXT PRIMARY KEY,
  room_code TEXT REFERENCES rooms(code) ON DELETE CASCADE,
  value TEXT NOT NULL,
  timestamp INT8 NOT NULL
);

-- Enable Realtime
ALTER PUBLICATION supabase_realtime ADD TABLE rooms, songs, vibes;
ALTER TABLE rooms ENABLE ROW LEVEL SECURITY;
ALTER TABLE songs ENABLE ROW LEVEL SECURITY;
ALTER TABLE vibes ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public full access on rooms" ON rooms FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public full access on songs" ON songs FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public full access on vibes" ON vibes FOR ALL USING (true) WITH CHECK (true);
