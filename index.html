    Database schema copy SQL 
    create table if not exists rooms (
  id uuid primary key default gen_random_uuid(),
  code text unique not null,
  created_at timestamptz default now()
);
create table if not exists songs (
  id uuid primary key default gen_random_uuid(),
  room_code text not null,
  title text not null,
  votes int4 not null default 0,
  created_at timestamptz default now()
);
create table if not exists vibes (
  id uuid primary key default gen_random_uuid(),
  room_code text not null,
  value text not null,
  created_at timestamptz default now()
);

alter table rooms  enable row level security;
alter table songs  enable row level security;
alter table vibes  enable row level security;
create policy "anon all rooms" on rooms for all using (true) with check (true);
create policy "anon all songs" on songs for all using (true) with check (true);
create policy "anon all vibes" on vibes for all using (true) with check (true);

-- atomic upvote so simultaneous voters never overwrite each other
create or replace function bump_vote(song_id uuid)
returns void language sql as $$
  update songs set votes = votes + 1 where id = song_id;
$$;

alter publication supabase_realtime add table rooms, songs, vibes;
